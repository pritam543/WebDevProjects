⚡ JavaScript (Logic & LocalStorage)
let transactions = JSON.parse(localStorage.getItem("expense_transactions")) || [];

function updateValues() {
    const amounts = transactions.map(t => t.amount);
    const total = amounts.reduce((acc, item) => acc + item, 0).toFixed(2);
    const income = amounts.filter(item => item > 0).reduce((acc, item) => acc + item, 0).toFixed(2);
    const expense = (amounts.filter(item => item < 0).reduce((acc, item) => acc + item, 0) * -1).toFixed(2);

    document.getElementById("totalBalance").textContent = `₹${total}`;
    document.getElementById("totalIncome").textContent = `+₹${income}`;
    document.getElementById("totalExpense").textContent = `-₹${expense}`;
}

function addTransaction(e) {
    e.preventDefault();
    const text = document.getElementById("text").value.trim();
    const amount = Number(document.getElementById("amount").value);
    transactions.push({ id: Date.now(), text, amount });
    localStorage.setItem("expense_transactions", JSON.stringify(transactions));
    renderTransactions();
    updateValues();
}
