⚡ JavaScript (Logic)
let startTime = 0, elapsedTime = 0, timerInterval = null, lapCounter = 1;

function startTimer() {
    startTime = Date.now() - elapsedTime;
    timerInterval = setInterval(() => {
        elapsedTime = Date.now() - startTime;
        printTime();
    }, 10);
    document.getElementById("startBtn").disabled = true;
    document.getElementById("pauseBtn").disabled = false;
    document.getElementById("lapBtn").disabled = false;
}

function pauseTimer() {
    clearInterval(timerInterval);
    document.getElementById("startBtn").disabled = false;
    document.getElementById("pauseBtn").disabled = true;
    document.getElementById("lapBtn").disabled = true;
}

function resetTimer() {
    clearInterval(timerInterval);
    elapsedTime = 0;
    lapCounter = 1;
    printTime();
    document.getElementById("lapsList").innerHTML = "";
    document.getElementById("startBtn").disabled = false;
    document.getElementById("pauseBtn").disabled = true;
    document.getElementById("lapBtn").disabled = true;
}

function recordLap() {
    const formatted = formatTime(elapsedTime);
    const lapTimeStr = `${formatted.m}:${formatted.s}:${formatted.ms}`;
    const li = document.createElement("li");
    li.innerHTML = `<span>Lap ${lapCounter}</span><strong>${lapTimeStr}</strong>`;
    document.getElementById("lapsList").prepend(li);
    lapCounter++;
}
